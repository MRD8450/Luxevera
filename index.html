function startEdit(id) { const cached = localStorage.getItem('luxevera_local_cache'); if (cached) { products = JSON.parse(cached); } const p = products.find(item => String(item.id || item.ID || item.Id) === String(id)); if (!p) { alert("Fehler: Produkt mit ID " + id + " wurde nicht gefunden."); return; } // Alle denkbaren Schreibweisen auslesen (sicherer Fallback) const prodName = p.name || p.Name || p.TITLE || p.title || ''; const prodPrice = p.price || p.Price || p.preis || p.Preis || ''; const prodCategory = p.category || p.Category || p.kategorie || p.Kategorie || 'parfum'; const prodDesc = p.description || p.Description || p.desc || p.Desc || p.beschreibung || p.Beschreibung || ''; document.getElementById('edit-id').value = p.id || p.ID || id; document.getElementById('p-name').value = prodName; document.getElementById('p-price').value = String(prodPrice).replace('€', '').trim(); document.getElementById('p-category').value = prodCategory; document.getElementById('p-desc').value = prodDesc; document.getElementById('p-image').value = ""; document.getElementById('form-title').innerText = "Produkt bearbeiten"; document.getElementById('save-btn').innerText = "Änderungen speichern"; document.getElementById('cancel-edit-btn').style.display = "inline-block"; document.getElementById('admin-panel').scrollIntoView({ behavior: 'smooth' }); }
